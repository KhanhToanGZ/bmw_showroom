// 5. Section 6: Configurator Calculation Logic
let basePrice = 3800000000;
let wheelPrice = 0;
let interiorPrice = 0;
let optionsPrice = 0;

function updateTotalPrice() {
    const total = basePrice + wheelPrice + interiorPrice + optionsPrice;
    const priceElem = document.getElementById('total-price');
    if (priceElem) {
        priceElem.innerText = total.toLocaleString('vi-VN') + " VNĐ";
    }
}

function setCarColor(name, hex) {
    const carImg = document.getElementById('config-car-img');
    if (!carImg) return;
    
    if (name === 'black') carImg.style.filter = 'brightness(0.65) contrast(1.2)';
    if (name === 'white') carImg.style.filter = 'brightness(1.25) contrast(0.95)';
    if (name === 'blue') carImg.style.filter = 'hue-rotate(180deg) brightness(0.9)';
    if (name === 'red') carImg.style.filter = 'hue-rotate(300deg) brightness(0.9)';
    if (name === 'verde') carImg.style.filter = 'hue-rotate(90deg) brightness(0.8)';
}

function setWheel(size, price, btn) {
    wheelPrice = price;
    document.querySelectorAll('.wheel-btn').forEach(b => {
        b.classList.remove('border-amber-400');
        b.classList.add('border-white/10');
    });
    if (btn) btn.classList.add('border-amber-400');
    updateTotalPrice();
}

function setInterior(name, price, btn) {
    interiorPrice = price;
    document.querySelectorAll('.interior-btn').forEach(b => {
        b.classList.remove('border-amber-400');
        b.classList.add('border-white/10');
    });
    if (btn) btn.classList.add('border-amber-400');
    updateTotalPrice();
}

function toggleOption(price, element) {
    if (element.checked) {
        optionsPrice += price;
    } else {
        optionsPrice -= price;
    }
    updateTotalPrice();
}
